\begin{figure}[htbp]

        \draw[->] (0,0) -- (0,4) node[above] {Pressure};
        \draw[thick,blue] plot[smooth] coordinates {(0,3.5) (1,3.2) (2,2.8) (3,2.3) (4,1.7) (5,1.0)};
    \end{tikzpicture}
